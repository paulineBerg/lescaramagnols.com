📁 Gestion des fichiers et dossiers
touch fichier.txt               : crée un fichier vide.
mkdir mon_dossier               : crée un dossier.
rm fichier.txt                  : supprime un fichier.
rm -r mon_dossier               : supprime un dossier et son contenu.
cp source.txt destination.txt   : copie un fichier.
mv ancien.txt nouveau.txt       : déplace ou renomme un fichier

---------------
fichier vite.config.mjs :
import { defineConfig } from 'vite';
import path from 'path';
import { viteStaticCopy } from 'vite-plugin-static-copy';
export default defineConfig({
  base: '/',
  build: {
    outDir: 'dist',
    emptyOutDir: true,
    manifest: true,
    rollupOptions: {
      input: path.resolve(__dirname, 'src/main.js'),
      output: {
        assetFileNames: ({ name }) => {
          if (!name) return 'assets/[name].[hash][extname]';
          const parts = name.split('/');
          const idx = parts.indexOf('assets');
          const subPath = idx >= 0 ? parts.slice(idx + 1, -1).join('/') : '';
          return subPath
            ? `assets/${subPath}/[name].[hash][extname]`
            : `assets/[name].[hash][extname]`;
        },
        entryFileNames: 'assets/[name].[hash].js'
      }
    },
    cssCodeSplit: true,
    minify: true
  },
  resolve: {
    alias: {
      '@': path.resolve(__dirname, 'src'),
      '~assets': path.resolve(__dirname, 'assets')
    }
  },
  css: {
    preprocessorOptions: {
      scss: {}
    }
  },
  plugins: [
    viteStaticCopy({
      targets: [
        {
          src: 'assets/images',
          dest: 'assets',
          keepHierarchyStructure: true
        },
        {
          src: 'assets/pdf',
          dest: 'assets',
          keepHierarchyStructure: true
        }
      ]
    })
  ]
});
